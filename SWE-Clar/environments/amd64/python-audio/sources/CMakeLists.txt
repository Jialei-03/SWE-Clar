cmake_minimum_required(VERSION 3.25)
project(ClarAudio VERSION 1.0.0 LANGUAGES CXX)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
add_subdirectory(/sdk sdk)
smtg_add_vst3plugin(ClarPiano instrument.cpp /sdk/public.sdk/source/main/linuxmain.cpp)
target_compile_features(ClarPiano PRIVATE cxx_std_17)
target_link_libraries(ClarPiano PRIVATE sdk)
add_executable(clar_render host.cpp /sdk/public.sdk/source/vst/hosting/module_linux.cpp /sdk/public.sdk/source/vst/hosting/plugprovider.cpp)
target_compile_features(clar_render PRIVATE cxx_std_17)
target_link_libraries(clar_render PRIVATE sdk_hosting ${CMAKE_DL_LIBS} pthread)
